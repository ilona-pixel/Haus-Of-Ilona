 // --- Scroll-triggered animations ---
  const scrollTriggers = document.querySelectorAll('.js-scroll-trigger');
  if (scrollTriggers.length > 0) {
    const scrollObserver = new IntersectionObserver((entries, observer) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          const target = entry.target as HTMLElement;
          const delay = target.dataset.delay;
          if (delay) {
            target.style.animationDelay = `${delay}s`;
          }
          target.classList.add('animate-fade-in-up');
          observer.unobserve(target);
        }
      });
    }, { threshold: 0.1 });
    scrollTriggers.forEach(trigger => scrollObserver.observe(trigger));
  }

  // --- VSL Modal Logic ---
  const vslTrigger = document.getElementById('vsl-trigger');
  const vslModal = document.getElementById('vsl-modal');
  const vslModalContent = document.getElementById('vsl-modal-content');
  const vslClose = document.getElementById('vsl-close');
  const vslIframe = document.getElementById('vsl-iframe') as HTMLIFrameElement;

  if (vslTrigger && vslModal && vslModalContent && vslClose && vslIframe) {
    const videoSrc = vslIframe.dataset.src;

    const openModal = () => {
      if (videoSrc) vslIframe.src = videoSrc;
      vslModal.classList.remove('opacity-0', 'pointer-events-none');
      vslModalContent.classList.remove('scale-95');
      document.body.style.overflow = 'hidden'; // Prevent background scrolling
    };

    const closeModal = () => {
      vslIframe.src = ''; // This stops the video
      vslModal.classList.add('opacity-0', 'pointer-events-none');
      vslModalContent.classList.add('scale-95');
      document.body.style.overflow = ''; // Restore scrolling
    };

    vslTrigger.addEventListener('click', openModal);
    vslClose.addEventListener('click', closeModal);

    // Close modal by clicking on the backdrop
    vslModal.addEventListener('click', (e) => {
      if (e.target === vslModal) {
        closeModal();
      }
    });
    
    // Close modal with Escape key
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && !vslModal.classList.contains('pointer-events-none')) {
        closeModal();
      }
    });
  }
});
